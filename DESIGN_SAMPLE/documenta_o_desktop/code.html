Component Construction

terminal Button.tsx 
// Strict adherence to the flat, technical design system
import React from 'react';
import { clsx } from 'clsx';

interface ActionProps {
  intent: 'primary' | 'ghost';
  label: string;
  icon?: string;
}

export const SystemAction = ({ intent, label, icon }: ActionProps) => {
  const baseStyle = 'font-label-caps text-label-caps uppercase px-4 py-2 rounded flex items-center gap-2 border transition-all duration-200';
  
  const styles = clsx(baseStyle, {
    'bg-primary-fixed-dim border-primary-fixed-dim text-on-primary-fixed hover:bg-primary focus:shadow-[0_0_10px_rgba(0,220,229,0.3)]': intent === 'primary',
    'bg-transparent border-border-subtle text-on-surface hover:border-primary-fixed-dim hover:text-primary-fixed-dim': intent === 'ghost',
  });

  return (
    <button className={styles}>
      {icon && <span className="material-symbols-outlined text-[16px]">{icon}</span>}
      <span>{label}</span>
    </button>
  );
};

Note the utilization of `font-label-caps` for button typography and the explicit outer bloom shadow applied via tailwind arbitrary values on focus to simulate an active glow without structural expansion.

Deployment Pipeline

  1. Pre-flight Checks: Ensure all JSON structural components align with the `style_` mandates.
  2. Asset Compilation: Tailwind CSS CLI runs in strict mode purging unused classes.
  3. Staging Deployment: Vercel trigger deploys to `staging.portfolio.dev`.
  4. Manual Review: Verify typography scale on mobile (`-mobile` variants).
